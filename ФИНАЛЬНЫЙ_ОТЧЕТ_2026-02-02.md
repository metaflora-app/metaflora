## 📝 Обязательно выполнить

### SQL миграция в Supabase
**Файл**: `metaflora-service/LABA_NOTIFICATION_SETTINGS_MIGRATION.sql`

```sql
DROP TABLE IF EXISTS laba_notification_settings CASCADE;

CREATE TABLE laba_notification_settings (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  tracked_account_id UUID NOT NULL,
  notifications_enabled BOOLEAN DEFAULT true,
  notification_time TIME DEFAULT '08:00:00',
  timezone TEXT DEFAULT 'Europe/Moscow',
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(user_id, tracked_account_id),
  FOREIGN KEY (tracked_account_id) REFERENCES laba_tracked_accounts(id) ON DELETE CASCADE
);

CREATE INDEX idx_notification_settings_user_id ON laba_notification_settings(user_id);
CREATE INDEX idx_notification_settings_tracked_account_id ON laba_notification_settings(tracked_account_id);
CREATE INDEX idx_notification_settings_enabled ON laba_notification_settings(notifications_enabled);

ALTER TABLE laba_notification_settings DISABLE ROW LEVEL SECURITY;
```
