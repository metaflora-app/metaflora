**Было (sendMessage):**


**Стало (sendDocument):**
```typescript
// Заголовок
await fetch(`https://api.telegram.org/bot${botToken}/sendMessage`, {
  body: JSON.stringify({
    chat_id: userId,
    text: `📚 *материалы урока: ${lessonTitle.toLowerCase()}*\n\n📎 отправляю файлы...`,
    parse_mode: 'Markdown'
  })
});

// Файлы
for (const material of materials) {
  await fetch(`https://api.telegram.org/bot${botToken}/sendDocument`, {
    body: JSON.stringify({
      chat_id: userId,
      document: material.url,
      caption: `📄 ${material.name}\n💾 ${material.size || ''}`,
      parse_mode: 'Markdown'
    })
  });
}
```
