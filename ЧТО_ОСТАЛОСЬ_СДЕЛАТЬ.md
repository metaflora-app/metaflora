```typescript
// lib/kinescope.ts
export async function uploadToKinescope(
  videoFile: File,
  title: string
): Promise<string> {
  const formData = new FormData();
  formData.append('file', videoFile);
  formData.append('title', title);
  
  const response = await fetch('https://api.kinescope.io/v1/videos', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer e7dc4869-562f-492a-811b-506296b20fb7`,
    },
    body: formData,
  });
  
  const data = await response.json();
  return data.id; // video_id
}
```
