## 🔧 ТЕХНИЧЕСКИЕ ДЕТАЛИ

### Поиск в мини-аппе:
```typescript
// При нажатии Enter
onKeyDown={(e) => {
  if (e.key === 'Enter') {
    searchInputRef.current?.blur(); // Скрыть клавиатуру
    handleSearch(); // Выполнить поиск
  }
}}

// Функция поиска
const handleSearch = () => {
  const filtered = allArticles.filter(article => {
    const searchableText = [
      article.title,
      article.annotation || '',
      ...(article.keywords || [])
    ].join(' ').toLowerCase();
    
    return searchWords.every(word => searchableText.includes(word));
  });
  
  if (filtered.length === 0) {
    // Telegram алерт
    telegram.WebApp.showPopup({
      message: 'статья не найдена. проверьте корректность написания'
    });
    // Очистить и вернуть все статьи
    setSearchValue('');
    setArticles(allArticles);
  }
};
```

```typescript
const handleDelete = async (id: string, title: string) => {
  if (!confirm(`Удалить статью "${title}"?`)) return;
  
  const response = await fetch(`/api/content/polygon-articles/${id}`, {
    method: 'DELETE',
  });
  
  if (response.ok) {
    alert('Статья удалена!');
    loadArticles(); // Перезагрузка списка
  }
};
```
